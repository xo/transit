import { readFile } from "node:fs/promises";

// Shapes by name.
const SHAPES = new Map([["square", 4], ["triangle", 3]]);

export class Shape {
  #name;
  constructor(name) {
    this.#name = name;
  }
  get sides() {
    return SHAPES.get(this.#name) ?? 0;
  }
}

export async function load(path, { strict = false } = {}) {
  const text = await readFile(path, "utf8");
  const names = text.split(/\s+/).filter(Boolean);
  if (strict && names.length === 0) throw new Error(`empty: ${path}`);
  return names.map((n) => new Shape(n));
}
