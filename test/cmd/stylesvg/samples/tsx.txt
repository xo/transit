import { useState } from "react";

// A list of shapes.
type Props = { names: string[]; title?: string };

export function Shapes({ names, title = "Shapes" }: Props) {
  const [count, setCount] = useState<number>(0);
  return (
    <section className="shapes">
      <h1>{title}</h1>
      <ul>
        {names.map((n) => (
          <li key={n} onClick={() => setCount(count + 1)}>
            {n}
          </li>
        ))}
      </ul>
    </section>
  );
}
