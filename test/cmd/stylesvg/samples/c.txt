#include <stdio.h>
#define MAX 16

/* A point on the plane. */
typedef struct {
	int x, y;
} point;

static int sum(const point *p, size_t n) {
	int total = 0;
	for (size_t i = 0; i < n; i++) {
		total += p[i].x * p[i].y; // area
	}
	return total;
}

int main(void) {
	point pts[MAX] = {{1, 2}, {3, 4}};
	printf("%d\n", sum(pts, 2));
	return 0;
}
