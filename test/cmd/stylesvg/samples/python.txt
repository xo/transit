"""Measure shapes."""
from dataclasses import dataclass
import math


@dataclass(frozen=True)
class Circle:
    radius: float = 1.0

    def area(self) -> float:
        return math.pi * self.radius ** 2


def total(*shapes: Circle, strict: bool = False) -> float:
    # Sum the areas.
    if strict and not shapes:
        raise ValueError(f"no shapes: {len(shapes)}")
    return sum(s.area() for s in shapes)
