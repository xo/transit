begin
  declare or replace variable n int default 3;
  declare bad_div condition for sqlstate '22012';
  declare exit handler for sqlexception, not found
    begin
      set n = -1;
    end;
  declare continue handler for bad_div set n = 0;
  outer_loop: while n > 0 do
    set n = n - 1;
    iterate outer_loop;
  end while outer_loop;
  for r as select id from t do
    leave;
  end for;
  signal sqlstate '45000' set message_text = 'stop';
end;
