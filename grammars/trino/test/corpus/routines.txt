WITH FUNCTION fib(n bigint) RETURNS bigint
BEGIN
  DECLARE a, b bigint DEFAULT 1;
  DECLARE c bigint;
  IF n <= 2 THEN
    RETURN 1;
  ELSEIF n = 3 THEN
    RETURN 2;
  ELSE
    SET c = 0;
  END IF;
  RETURN a + b;
END
SELECT fib(10);

WITH FUNCTION count_to(n bigint) RETURNS bigint
BEGIN
  DECLARE i bigint DEFAULT 0;
  top: LOOP
    SET i = i + 1;
    IF i >= n THEN
      LEAVE top;
    END IF;
    ITERATE top;
  END LOOP;
  w: WHILE i > 0 DO
    SET i = i - 1;
  END WHILE;
  REPEAT
    SET i = i + 2;
  UNTIL i > 10 END REPEAT;
  RETURN i;
END
SELECT count_to(5);

WITH FUNCTION grade(s bigint) RETURNS varchar
BEGIN
  CASE s
    WHEN 1 THEN RETURN 'one';
    ELSE RETURN 'many';
  END CASE;
  CASE WHEN s > 10 THEN RETURN 'big'; END CASE;
  RETURN NULL;
END
SELECT grade(3);
