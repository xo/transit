CREATE OR REPLACE FUNCTION fib(n bigint)
RETURNS bigint
DETERMINISTIC
COMMENT 'the Fibonacci number of n'
BEGIN
  DECLARE a, b bigint DEFAULT 1;
  DECLARE c bigint;
  IF n <= 2 THEN
    RETURN 1;
  END IF;
  WHILE n > 2 DO
    SET n = n - 1;
    SET c = a + b;
    SET a = b;
    SET b = c;
  END WHILE;
  RETURN c;
END;
